// __tests__/offline-sync.test.js

import { queueMutation, flushQueue, getQueuedMutations, clearQueue } from '../lib/offline';

// Mock fetch globally
global.fetch = jest.fn();

// Helper to control navigator.onLine
function setOnline(value) {
  Object.defineProperty(window.navigator, 'onLine', {
    configurable: true,
    value,
  });
}

describe('Offline sync queue', () => {
  beforeEach(() => {
    fetch.mockClear();
    clearQueue(); // assumes a way to reset IndexedDB/queue state between tests
  });

  test('queues a mutation instead of hitting the network when offline', async () => {
    setOnline(false);

    await queueMutation({
      type: 'CREATE_INVOICE',
      payload: { customerId: 'c1', amount: 500 },
    });

    const queued = await getQueuedMutations();
    expect(queued).toHaveLength(1);
    expect(queued[0].payload.amount).toBe(500);
    expect(fetch).not.toHaveBeenCalled();
  });

  test('flushes queued mutations in order once back online', async () => {
    setOnline(false);
    await queueMutation({ type: 'CREATE_INVOICE', payload: { id: 1 } });
    await queueMutation({ type: 'UPDATE_INVOICE', payload: { id: 1, status: 'paid' } });

    setOnline(true);
    fetch.mockResolvedValue({ ok: true, json: async () => ({ success: true }) });

    await flushQueue();

    expect(fetch).toHaveBeenCalledTimes(2);
    // Order matters — first call should be the CREATE, second the UPDATE
    expect(fetch.mock.calls[0][1].body).toContain('CREATE_INVOICE');
    expect(fetch.mock.calls[1][1].body).toContain('UPDATE_INVOICE');
  });

  test('retries only the failed mutation, not the whole queue', async () => {
    setOnline(false);
    await queueMutation({ type: 'CREATE_INVOICE', payload: { id: 1 } });
    await queueMutation({ type: 'CREATE_INVOICE', payload: { id: 2 } });

    setOnline(true);
    fetch
      .mockResolvedValueOnce({ ok: false, status: 500 }) // first fails
      .mockResolvedValueOnce({ ok: true, json: async () => ({ success: true }) }); // second succeeds

    await flushQueue();

    const remaining = await getQueuedMutations();
    // Only the failed mutation should still be in the queue
    expect(remaining).toHaveLength(1);
    expect(remaining[0].payload.id).toBe(1);
  });

  test('editing the same record twice offline sends both mutations in order', async () => {
    setOnline(false);
    await queueMutation({ type: 'UPDATE_INVOICE', payload: { id: 1, amount: 500 } });
    await queueMutation({ type: 'UPDATE_INVOICE', payload: { id: 1, amount: 750 } });

    const queued = await getQueuedMutations();
    expect(queued).toHaveLength(2);
    expect(queued[0].payload.amount).toBe(500);
    expect(queued[1].payload.amount).toBe(750);
  });

  test('queue survives simulated reload (persisted, not just in-memory)', async () => {
    setOnline(false);
    await queueMutation({ type: 'CREATE_INVOICE', payload: { id: 1 } });

    // Simulate a fresh module load / page reload by re-reading straight from IndexedDB
    const persisted = await getQueuedMutations();
    expect(persisted).toHaveLength(1); // fails if queue lived only in a JS variable
  });
});

## Deployment



This is where the "local-first per vendor" vs "hosted multi-tenant" decision actually forks the architecture, so worth deciding explicitly rather than drifting into one:



**Option A — Local-first, self-hosted per vendor (matches your current architecture closest)**

- Each vendor runs their own instance — could be as simple as a Raspberry Pi or a cheap Android tablet running a local Node server, or even fully client-side if you lean harder into IndexedDB as the source of truth.

- Pros: no per-vendor cloud cost, no cross-vendor data exposure risk, works with your offline-first design philosophy natively.

- Cons: you (or the vendor) become responsible for backups — a lost/broken device means lost udhar records unless you add periodic export/backup. Also harder to support/debug remotely.



**Option B — Hosted multi-tenant (Vercel/Railway/Render + a real DB)**

- One deployment, vendor data isolated by tenant ID. This is the more "SaaS" path.

- Requires the JSON-file `db.js` to actually become a real DB (Postgres/SQLite via Prisma or similar) — flat JSON files don't work with concurrent multi-tenant writes.

- The offline-first PWA logic mostly still works the same way (client syncs to your hosted API instead of a local one) — that part doesn't need to change.

- Pros: centralized backups, easier updates/support, you could add vendor-to-vendor features (the "Nearby Vendor Network" idea from earlier) since data's in one place.

- Cons: real infra cost, and you take on responsibility for vendors' financial data security (auth hardening becomes non-optional, not just nice-to-have).



**My honest read:** given the offline-first design is already this far along and the target users are individual street vendors/MSMEs (not enterprises), Option A with an added "export/backup to cloud" safety net is probably the more honest fit — you get hosted-style backup without taking on multi-tenant auth/security complexity you don't need yet. You can always migrate to B later if you get traction and vendors start asking for cross-device sync.



Which direction are you leaning, or is this still open? That'll decide whether "next" is a DB migration (Option B) or a backup/export mechanism (Option A).

  const containerRef = useRef<HTMLDivElement>(null)
  const sceneRef = useRef<{
    camera: any
    scene: any
    renderer: any

    script.onload = () => {
      if (containerRef.current && window.THREE) {

      }
    }
    document.head.appendChild(script)

    return () => {
      // Cleanup
      if (sceneRef.current.animationId) {
        cancelAnimationFrame(sceneRef.current.animationId)
      }
      if (sceneRef.current.renderer) {

    const THREE = window.THREE
    const container = containerRef.current

    // Clear any existing content
    container.innerHTML = ""

    // Initialize camera
    const camera = new THREE.Camera()
    camera.position.z = 1

    // Initialize scene
    const scene = new THREE.Scene()

    // Create geometry
    const geometry = new THREE.PlaneBufferGeometry(2, 2)

    // Define uniforms
    const uniforms = {
      time: { type: "f", value: 1.0 },
      resolution: { type: "v2", value: new THREE.Vector2() },
    }

    // Vertex shader
    const vertexShader = `
      void main() {
        gl_Position = vec4( position, 1.0 );
      }
    `

    // Fragment shader
    const fragmentShader = `
      #define TWO_PI 6.2831853072
      #define PI 3.14159265359

      precision highp float;
      uniform vec2 resolution;
      uniform float time;
        
      float random (in float x) {
          return fract(sin(x)*1e4);
      }
      float random (vec2 st) {
          return fract(sin(dot(st.xy,
                               vec2(12.9898,78.233)))*
              43758.5453123);
      }
      
      varying vec2 vUv;

      void main(void) {
        vec2 uv = (gl_FragCoord.xy * 2.0 - resolution.xy) / min(resolution.x, resolution.y);
        
        vec2 fMosaicScal = vec2(4.0, 2.0);
        vec2 vScreenSize = vec2(256,256);
        uv.x = floor(uv.x * vScreenSize.x / fMosaicScal.x) / (vScreenSize.x / fMosaicScal.x);
        uv.y = floor(uv.y * vScreenSize.y / fMosaicScal.y) / (vScreenSize.y / fMosaicScal.y);

          }
        }

        gl_FragColor = vec4(color[2],color[1],color[0],1.0);
      }

      vertexShader: vertexShader,
